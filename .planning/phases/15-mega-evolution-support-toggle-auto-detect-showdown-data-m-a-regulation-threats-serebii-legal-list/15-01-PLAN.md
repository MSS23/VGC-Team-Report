**1. Create `src/lib/utils/mega-resolver.ts` (NEW FILE):**

```typescript
import { MEGA_POKEMON_LIST } from "@/lib/data/mega-pokemon";
import type { MegaPokemonEntry } from "@/lib/data/mega-pokemon";
import { POKEMON_DATA } from "@/lib/data/pokemon";
import type { ParsedPokemon, PokemonData, StatSpread } from "@/lib/types/pokemon";
import { NATURES } from "@/lib/data/natures";

export interface MegaResolution {
  isMega: boolean;
  displaySpecies: string;        // species key for sprite lookup
  effectiveData: PokemonData | null;  // types/stats/ability to render
  megaEntry: MegaPokemonEntry | null;
  canToggle: boolean;            // true when a mega form exists for this pokemon
  effectiveCalculatedStats: StatSpread | null; // recalculated stats with mega base stats
}

export function resolveMegaForm(
  parsed: ParsedPokemon,
  megaStateOverride: boolean | undefined,
  regulation: string | undefined,
): MegaResolution {
  const baseData = POKEMON_DATA[parsed.species.toLowerCase()] ?? null;
  const regAllowed = !regulation || regulation === "Reg M-A";
  const megaEntry = findMegaByItem(parsed.species, parsed.item?.toLowerCase() ?? "");
  const canToggle = regAllowed && megaEntry !== null;

  let isMega: boolean;
  if (!canToggle) {
    isMega = false;
  } else if (megaStateOverride !== undefined) {
    isMega = megaStateOverride;
  } else {
    // Auto-detect: mega stone equipped = show mega
    isMega = megaEntry !== null;
  }

  if (isMega && megaEntry) {
    const megaData = POKEMON_DATA[megaEntry.dataKey] ?? null;
    const effectiveCalculatedStats = megaData
      ? calculateStats(megaData.baseStats, parsed.evs, parsed.ivs, parsed.nature, parsed.level)
      : null;
    return {
      isMega: true,
      displaySpecies: megaEntry.dataKey,
      effectiveData: megaData,
      megaEntry,
      canToggle,
      effectiveCalculatedStats,
    };
  }

  return {
    isMega: false,
    displaySpecies: parsed.species,
    effectiveData: baseData,
    megaEntry,
    canToggle,
    effectiveCalculatedStats: null, // use original calculatedStats
  };
}

function findMegaByItem(species: string, itemLower: string): MegaPokemonEntry | null {
  if (!itemLower) return null;
  const speciesLower = species.toLowerCase();
  for (const entry of MEGA_POKEMON_LIST) {
    if (
      entry.baseName.toLowerCase() === speciesLower &&
      entry.megaStone.toLowerCase() === itemLower
    ) {
      return entry;
    }
  }
  return null;
}

function calculateStats(
  base: StatSpread,
  evs: StatSpread,
  ivs: StatSpread,
  nature: string,
  level: number,
): StatSpread {
  const natureData = NATURES[nature];
  const stats = {} as StatSpread;
  for (const stat of ["hp", "atk", "def", "spa", "spd", "spe"] as const) {
    if (stat === "hp") {
      stats.hp = Math.floor(((2 * base.hp + ivs.hp + Math.floor(evs.hp / 4)) * level) / 100) + level + 10;
    } else {
      let natureMult = 1;
      if (natureData?.plus === stat) natureMult = 1.1;
      else if (natureData?.minus === stat) natureMult = 0.9;
      stats[stat] = Math.floor((Math.floor(((2 * base[stat] + ivs[stat] + Math.floor(evs[stat] / 4)) * level) / 100) + 5) * natureMult);
    }
  }
  return stats;
}
```

Key points:
- `findMegaByItem` matches BOTH `baseName` AND `megaStone` — this correctly handles Charizard having two megas (Charizardite X vs Charizardite Y resolve to different entries).
- `calculateStats` recomputes level-50 stats with mega base stats using the same formula as the game engine.
- `regulation` gating: only "Reg M-A" or unset allows mega toggle.
- Returns `effectiveCalculatedStats` so consumers can swap in mega stats without re-running the analysis pipeline.
