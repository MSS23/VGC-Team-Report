## DRAFT 4 — BreadcrumbList helper + per-page emission

**File A:** `src/components/seo/JsonLd.tsx` (add to existing file)

```tsx
export interface Breadcrumb {
  name: string;
  url: string;
}

/**
 * Emit a Schema.org BreadcrumbList. Pass an ordered array starting with the
 * top-level node (e.g. Home) and ending with the current page. Each url must
 * be absolute (https://pokemonvgcteamreport.com/...).
 */
export function BreadcrumbListJsonLd({ items }: { items: Breadcrumb[] }) {
  if (items.length < 2) return null; // single-item lists are meaningless
  return (
    <JsonLd
      data={{
        "@context": "https://schema.org",
        "@type": "BreadcrumbList",
        itemListElement: items.map((it, i) => ({
          "@type": "ListItem",
          position: i + 1,
          name: it.name,
          item: it.url,
        })),
      }}
    />
  );
}
```
